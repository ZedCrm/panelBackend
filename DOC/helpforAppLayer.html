📌 لایه App دقیقاً چیه؟

لایه App جاییه که همه منطق‌های کسب و کار پروژه توش نوشته میشه.

این لایه وسط قرار داره: کنترلرها ازش استفاده می‌کنن و خودش از DbContext استفاده میکنه.

┌─────────────────┐
│   Controller    │  ← فقط درخواست میگیره و جواب میده
├─────────────────┤
│    لایه App     │  ← اینجا همه کارهای اصلی انجام میشه
├─────────────────┤
│   DbContext     │  ← ارتباط با دیتابیس
└─────────────────┘
💡 قانون طلایی: هیچ کد منطق کسب و کاری توی کنترلر ننویس!
کنترلر فقط باید BusinessService یا CrudService رو صدا بزنه.
⚙️ دو نوع سرویس در لایه App

🏗️ BaseService (پدر همه BusinessServiceها)

BaseService یه کلاس پایه است که همه BusinessServiceها ازش ارث می‌برن. متدهای پرکاربردی داره که کار رو خیلی راحت می‌کنه:

متدکاربردمثال
GetActiveQuery()گرفتن کوئری فقط از رکوردهای فعال (IsDeleted=false)نمایش لیست کاربران
ValidateModel(model)اعتبارسنجی خودکارچک کردن فرم ورودی
GetEntityOrNullAsync(predicate)گرفتن یک رکورد یا nullپیدا کردن کاربر با ایمیل
ExistsAsync(predicate)چک کردن وجود رکوردآیا این ایمیل قبلا ثبت شده؟
ExecuteAsync(action, msg)اجرا با مدیریت خطاذخیره کردن با try-catch
📝 نکته: وقتی BusinessService می‌سازی، کافیه از BaseService ارث ببری تا این متدها در اختیارت باشن.
🎯 مثال عملی: ساخت سرویس محصولات (Product)

فرض کن می‌خوای یه سرویس برای محصولات بنویسی که:

  • محصول جدید بسازه
  • کد محصول نباید تکراری باشه
  • قیمت محصول باید مثبت باشه
  • قابلیت جستجو بین محصولات داشته باشه

مرحله 1: ساخت BusinessService (برای منطق‌های خاص)

// App/Object/Shop/ProductBusinessService.cs
using Microsoft.EntityFrameworkCore;
using MyFrameWork.AppTool;
using MyFrameWork.AppTool.ResultType;
using ConfApp;
using Domain.Objects.Shop;

namespace App.Object.Shop
{
    public class ProductBusinessService : BaseService<Product>
    {
        public ProductBusinessService(MyContext context) : base(context)
        {
        }

        // 1. بررسی یکتایی کد محصول
        public async Task<StatusResult> IsProductCodeUniqueAsync(string productCode, int? excludeId = null)
        {
            var query = GetActiveQuery();  // فقط محصولات فعال
            
            if (excludeId.HasValue)
                query = query.Where(p => p.Id != excludeId.Value);

            bool exists = await query.AnyAsync(p => p.ProductCode == productCode);
            
            if (exists)
                return ResultFactory.Status(ResultStatusEnum.Conflict, 
                    "کد محصول تکراری است.");

            return ResultFactory.Status(ResultStatusEnum.Success);
        }

        // 2. بررسی قیمت (با استفاده از متد BaseService)
        public StatusResult ValidatePrice(int price)
        {
            if (price <= 0)
                return ResultFactory.Status(ResultStatusEnum.ValidationFailed, 
                    "قیمت باید بزرگتر از صفر باشد.");
            return ResultFactory.Status(ResultStatusEnum.Success);
        }

        // 3. جستجوی پیشرفته محصولات
        public async Task<ListDataResult<ProductView>> SearchProductsAsync(
            string? searchTerm, int? minPrice, int? maxPrice, Pagination pagination)
        {
            var query = GetActiveQuery()  // فقط فعال‌ها
                .Include(p => p.CountType) // همراه با واحد شمارش
                .AsQueryable();

            // فیلتر بر اساس نام
            if (!string.IsNullOrEmpty(searchTerm))
                query = query.Where(p => p.Name.Contains(searchTerm));

            // فیلتر قیمت
            if (minPrice.HasValue)
                query = query.Where(p => p.Price >= minPrice.Value);
            if (maxPrice.HasValue)
                query = query.Where(p => p.Price <= maxPrice.Value);

            int total = await query.CountAsync();

            var products = await query
                .Skip(pagination.CalculateSkip())
                .Take(pagination.PageSize)
                .ToListAsync();

            var views = products.Select(p => new ProductView
            {
                Id = p.Id,
                ProductCode = p.ProductCode ?? "",
                Name = p.Name,
                Price = p.Price
            }).ToList();

            return ResultFactory.List(ResultStatusEnum.Success, views, total, pagination);
        }
    }
}

مرحله 2: ساخت CrudService (برای عملیات ساده)

// App/Object/Shop/ProductApp.cs
using AutoMapper;
using MyFrameWork.AppTool;
using MyFrameWork.AppTool.ResultType;
using ConfApp;
using Domain.Objects.Shop;

namespace App.Object.Shop
{
    public class ProductApp : CrudService<Product, ProductView, ProductCreate, ProductUpdate, int>
    {
        private readonly ProductBusinessService _productBusiness;

        public ProductApp(MyContext context, IMapper mapper, ProductBusinessService productBusiness) 
            : base(context, mapper)
        {
            _productBusiness = productBusiness;
        }

        // ساخت محصول جدید
        public override async Task<StatusResult> CreateAsync(ProductCreate dto)
        {
            // 1. اعتبارسنجی خودکار (با Attributes)
            var validation = ValidateModel(dto);
            if (!validation.IsSuccess) return validation;

            // 2. بررسی قیمت
            var priceCheck = _productBusiness.ValidatePrice(dto.Price);
            if (!priceCheck.IsSuccess) return priceCheck;

            // 3. بررسی یکتایی کد محصول
            var uniqueCheck = await _productBusiness.IsProductCodeUniqueAsync(dto.ProductCode);
            if (!uniqueCheck.IsSuccess) return uniqueCheck;

            // 4. ذخیره در دیتابیس
            var product = _mapper.Map<Product>(dto);
            await _dbSet.AddAsync(product);
            await _context.SaveChangesAsync();

            return ResultFactory.Status(ResultStatusEnum.Success, "محصول با موفقیت ایجاد شد.");
        }

        // جستجو (با استفاده از BusinessService)
        public async Task<ListDataResult<ProductView>> SearchAsync(
            string? searchTerm, int? minPrice, int? maxPrice, Pagination pagination)
        {
            return await _productBusiness.SearchProductsAsync(searchTerm, minPrice, maxPrice, pagination);
        }
    }
}

// App.Contracts/Object/Shop/ProductView.cs
using System.ComponentModel.DataAnnotations;

public class ProductView
{
    public int Id { get; set; }
    
    [Required(ErrorMessage = "کد محصول الزامی است.")]
    [StringLength(20, MinimumLength = 3)]
    public string ProductCode { get; set; } = string.Empty;
    
    [Required(ErrorMessage = "نام محصول الزامی است.")]
    [StringLength(50, MinimumLength = 2)]
    public string Name { get; set; } = string.Empty;
    
    [Required]
    [Range(1, 100000000, ErrorMessage = "قیمت باید بین 1 تا 100 میلیون باشد.")]
    public int Price { get; set; }
}

public class ProductCreate
{
    [Required]
    public string ProductCode { get; set; } = string.Empty;
    
    [Required]
    public string Name { get; set; } = string.Empty;
    
    [Required]
    public int Price { get; set; }
    
    [Required]
    public int CountTypeId { get; set; }
}

public class ProductUpdate : ProductCreate
{
    [Required]
    public int Id { get; set; }
}

1️⃣ ولیدیشن خودکار (با Attributes) - توسط ValidateModel
2️⃣ ولیدیشن قیمت - توسط متد ساده BusinessService
3️⃣ ولیدیشن یکتایی کد - توسط متد دیتابیسی BusinessService
🎮 کنترلر چطور کار میکنه؟
// API/Controllers/Shop/ProductController.cs
using Microsoft.AspNetCore.Mvc;

[ApiController]
[Route("api/[controller]")]
public class ProductController : ControllerBase
{
    private readonly ProductApp _productApp;
    private readonly ProductBusinessService _productBusiness;

    public ProductController(ProductApp productApp, ProductBusinessService productBusiness)
    {
        _productApp = productApp;
        _productBusiness = productBusiness;
    }

    [HttpPost("create")]
    public async Task<IActionResult> Create(ProductCreate dto)
    {
        // فقط صدا زدن سرویس - هیچ منطقی اینجا نیست!
        var result = await _productApp.CreateAsync(dto);
        
        if (!result.IsSuccess)
            return BadRequest(result);
        
        return Ok(result);
    }

    [HttpGet("search")]
    public async Task<IActionResult> Search([FromQuery] string? term, [FromQuery] int? minPrice, [FromQuery] int? maxPrice, [FromQuery] Pagination pagination)
    {
        var result = await _productApp.SearchAsync(term, minPrice, maxPrice, pagination);
        return Ok(result);
    }
}
📌 ببین چقدر کنترلر تمیز شده: فقط داره سرویس رو صدا میزنه و جواب میده. هیچ if و for و کوئری‌ای توش نیست!
🔧 متدهای BaseService در عمل

1. GetActiveQuery() - گرفتن فقط رکوردهای فعال

// مثال: گرفتن محصولات فعال با قیمت بالای 1000
var activeProducts = GetActiveQuery().Where(p => p.Price > 1000).ToList();

// معادل این کوئری SQL میشه:
// SELECT * FROM Products WHERE IsDeleted = 0 AND Price > 1000

2. ValidateModel() - اعتبارسنجی خودکار یک خطی

// بدرد میخوره وقتی میخوای DTO رو اعتبارسنجی کنی
var validation = ValidateModel(dto);
if (!validation.IsSuccess) return validation;
// دیگه نیازی نیست 10 خط کد validation بنویسی!

3. GetEntityOrNullAsync() - پیدا کردن یا null

// پیدا کردن محصول با شناسه
var product = await GetEntityOrNullAsync(p => p.Id == id);

if (product == null)
    return ResultFactory.Status(ResultStatusEnum.NotFound, "محصول یافت نشد.");

4. ExistsAsync() - چک کردن وجود

// چک کن ببین محصول با این کد وجود داره؟
bool exists = await ExistsAsync(p => p.ProductCode == code);

if (exists)
    return ResultFactory.Status(ResultStatusEnum.Conflict, "کد محصول تکراری است.");

5. ExecuteAsync() - اجرای امن با مدیریت خطا

// وقتی چندتا عملیات داری که ممکنه خطا بده
var result = await ExecuteAsync(async () =>
{
    await _dbSet.AddAsync(product);
    await _context.SaveChangesAsync();
}, "محصول با موفقیت ذخیره شد.");

// این متد خودکار خطاها رو میگیره و هندل میکنه
📋 خلاصه سریع (چیت‌شیت)
اگه بخوام...از چی استفاده کنم؟چطور؟
لیست همه رو بگیرمCrudServiceawait GetAllAsync(pagination)
یکی رو با شناسه بگیرمCrudServiceawait GetByIdAsync(id)
جدید بسازمCrudServiceoverride CreateAsync(dto)
بررسی کنم تکراری نباشهBusinessServiceawait IsUniqueAsync(value)
جستجوی پیشرفته کنمBusinessServiceکوئری بنویسم
فقط رکوردهای فعال ببینمBaseServiceGetActiveQuery()
اعتبارسنجی خودکار کنمBaseServiceValidateModel(dto)
با خطاهای دیتابیسی کار کنمBaseServiceExecuteAsync(action, msg)
⚠️ یادت نره:
  • همیشه از GetActiveQuery() استفاده کن تا رکوردهای حذف شده رو نبینی
  • ولیدیشن‌های ساده رو با ValidateModel() انجام بده
  • ولیدیشن دیتابیسی (مثل یکتایی) رو توی BusinessService بنویس
  • هیچ وقت توی کنترلر کوئری ننویس
✨ یک نمونه کامل: سرویس کاربر (User)
// App/Object/Base/Users/UserBusinessService.cs
public class UserBusinessService : BaseService<User>
{
    public UserBusinessService(MyContext context) : base(context) { }

    // بررسی یکتایی ایمیل
    public async Task<bool> IsEmailUniqueAsync(string email, int? excludeUserId = null)
    {
        var query = GetActiveQuery();
        if (excludeUserId.HasValue)
            query = query.Where(u => u.Id != excludeUserId.Value);
        return !await query.AnyAsync(u => u.Email == email);
    }

    // بررسی یکتایی نام کاربری
    public async Task<bool> IsUsernameUniqueAsync(string username, int? excludeUserId = null)
    {
        var query = GetActiveQuery();
        if (excludeUserId.HasValue)
            query = query.Where(u => u.Id != excludeUserId.Value);
        return !await query.AnyAsync(u => u.Username == username);
    }
}

// App/Object/Base/Users/UsersApp.cs
public class UsersApp : CrudService<User, UsersView, UsersCreat, UsersUpdate, int>, IUsersApp
{
    private readonly UserBusinessService _userBusiness;

    public UsersApp(MyContext context, IMapper mapper, UserBusinessService userBusiness) 
        : base(context, mapper)
    {
        _userBusiness = userBusiness;
    }

    public override async Task<StatusResult> CreateAsync(UsersCreat dto)
    {
        // 1. اعتبارسنجی خودکار
        var validation = ValidateModel(dto);
        if (!validation.IsSuccess) return validation;

        // 2. بررسی یکتایی ایمیل
        if (!await _userBusiness.IsEmailUniqueAsync(dto.Email))
            return ResultFactory.Status(ResultStatusEnum.Conflict, "ایمیل قبلاً ثبت شده است.");

        // 3. بررسی یکتایی نام کاربری
        if (!await _userBusiness.IsUsernameUniqueAsync(dto.Username))
            return ResultFactory.Status(ResultStatusEnum.Conflict, "نام کاربری قبلاً ثبت شده است.");

        // 4. ساخت کاربر
        var user = _mapper.Map<User>(dto);
        user.PasswordHash = BCrypt.Net.BCrypt.HashPassword(dto.Password);
        
        await _dbSet.AddAsync(user);
        await _context.SaveChangesAsync();

        return ResultFactory.Status(ResultStatusEnum.Success, "کاربر با موفقیت ایجاد شد.");
    }
}
❓ سوالات متداول
سوال: کی باید BusinessService بسازم و کی فقط از CrudService استفاده کنم؟
جواب: اگه عملیاتت فقط یه SaveChanges ساده هست، همون CrudService کافیه. اگه ولیدیشن دیتابیسی، شرط‌های پیچیده، یا جستجو داری، BusinessService بساز.
سوال: چرا نباید توی کنترلر کوئری بنویسم؟
جواب: چون اگه کوئری توی کنترلر باشه، تکرار میشه و نگهداریش سخت میشه. توی لایه App کوئری رو یه جا مینویسی و همه جا استفاده میکنی.
سوال: متد GetActiveQuery() چطور کار میکنه؟
جواب: یه IQueryable برمیگردونه که شرط IsDeleted == false روش زده شده. یعنی فقط رکوردهای فعال رو بهت نشون میده.
